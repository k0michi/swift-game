cmake_minimum_required(VERSION 3.24)
project(ExtractSDLTarget LANGUAGES NONE)

find_package(SDL3 CONFIG REQUIRED)

if(NOT DEFINED OUTPUT_DIRECTORY)
    message(FATAL_ERROR "OUTPUT_DIRECTORY is required")
endif()

get_target_property(sdl_library SDL3::SDL3-static IMPORTED_LOCATION_RELEASE)
if(NOT sdl_library)
    get_target_property(sdl_library SDL3::SDL3-static IMPORTED_LOCATION)
endif()
if(NOT sdl_library)
    message(FATAL_ERROR "Could not locate the SDL3 static library")
endif()

set(compiler_flags)
get_target_property(include_directories SDL3::Headers INTERFACE_INCLUDE_DIRECTORIES)
if(NOT "${include_directories}" STREQUAL "include_directories-NOTFOUND")
    foreach(directory IN LISTS include_directories)
        list(APPEND compiler_flags "-I${directory}")
    endforeach()
endif()

get_target_property(compile_definitions SDL3::Headers INTERFACE_COMPILE_DEFINITIONS)
if(NOT "${compile_definitions}" STREQUAL "compile_definitions-NOTFOUND")
    foreach(definition IN LISTS compile_definitions)
        list(APPEND compiler_flags "-D${definition}")
    endforeach()
endif()

set(linker_flags "${sdl_library}")
get_target_property(link_libraries SDL3::SDL3-static INTERFACE_LINK_LIBRARIES)
foreach(library IN LISTS link_libraries)
    if(library STREQUAL "SDL3::Headers")
        continue()
    elseif(library MATCHES "^\\$<LINK_ONLY:\\$<LINK_LIBRARY:FRAMEWORK,([^>]+)>>$")
        list(APPEND linker_flags "-Wl,-framework,${CMAKE_MATCH_1}")
    elseif(library MATCHES "^\\$<LINK_ONLY:\\$<LINK_LIBRARY:WEAK_FRAMEWORK,([^>]+)>>$")
        list(APPEND linker_flags "-Wl,-weak_framework,${CMAKE_MATCH_1}")
    elseif(library MATCHES "^\\$<LINK_ONLY:([^>]+)>$")
        set(library "${CMAKE_MATCH_1}")
        if(library MATCHES "^-" OR IS_ABSOLUTE "${library}")
            list(APPEND linker_flags "${library}")
        else()
            list(APPEND linker_flags "-l${library}")
        endif()
    elseif(library MATCHES "^-" OR IS_ABSOLUTE "${library}")
        list(APPEND linker_flags "${library}")
    elseif(NOT TARGET "${library}")
        list(APPEND linker_flags "-l${library}")
    endif()
endforeach()

get_target_property(link_options SDL3::SDL3-static INTERFACE_LINK_OPTIONS)
if(link_options)
    list(APPEND linker_flags ${link_options})
endif()

list(JOIN compiler_flags "\n" compiler_flags_text)
list(JOIN linker_flags "\n" linker_flags_text)
file(MAKE_DIRECTORY "${OUTPUT_DIRECTORY}")
file(WRITE "${OUTPUT_DIRECTORY}/compiler-flags.txt" "${compiler_flags_text}\n")
file(WRITE "${OUTPUT_DIRECTORY}/linker-flags.txt" "${linker_flags_text}\n")
