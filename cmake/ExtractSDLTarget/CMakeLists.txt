cmake_minimum_required(VERSION 3.24)
project(ExtractSDLTarget LANGUAGES NONE)

find_package(SDL3 CONFIG REQUIRED COMPONENTS SDL3-static)

if(NOT DEFINED OUTPUT_DIRECTORY)
    message(FATAL_ERROR "OUTPUT_DIRECTORY is required")
endif()

get_target_property(sdl_library SDL3::SDL3-static IMPORTED_LOCATION_RELEASE)
if(NOT sdl_library)
    get_target_property(sdl_library SDL3::SDL3-static IMPORTED_LOCATION)
endif()
if(NOT sdl_library)
    message(FATAL_ERROR "Could not locate the SDL3 static library")
endif()

set(compiler_flags)
get_target_property(include_directories SDL3::Headers INTERFACE_INCLUDE_DIRECTORIES)
if(NOT "${include_directories}" STREQUAL "include_directories-NOTFOUND")
    foreach(directory IN LISTS include_directories)
        list(APPEND compiler_flags "-I${directory}")
    endforeach()
endif()

get_target_property(compile_definitions SDL3::Headers INTERFACE_COMPILE_DEFINITIONS)
if(NOT "${compile_definitions}" STREQUAL "compile_definitions-NOTFOUND")
    foreach(definition IN LISTS compile_definitions)
        list(APPEND compiler_flags "-D${definition}")
    endforeach()
endif()

set_property(GLOBAL PROPERTY collected_linker_flags "${sdl_library}")
set_property(GLOBAL PROPERTY visited_link_targets "SDL3::SDL3-static")

function(append_link_flag flag)
    get_property(flags GLOBAL PROPERTY collected_linker_flags)
    list(APPEND flags "${flag}")
    set_property(GLOBAL PROPERTY collected_linker_flags "${flags}")
endfunction()

function(append_link_item item)
    if(item STREQUAL "" OR item STREQUAL "SDL3::Headers")
        return()
    endif()

    if(item MATCHES "^\\$<LINK_ONLY:\\$<LINK_LIBRARY:FRAMEWORK,([^>]+)>>$")
        append_link_flag("-Wl,-framework,${CMAKE_MATCH_1}")
        return()
    endif()

    if(item MATCHES "^\\$<LINK_ONLY:\\$<LINK_LIBRARY:WEAK_FRAMEWORK,([^>]+)>>$")
        append_link_flag("-Wl,-weak_framework,${CMAKE_MATCH_1}")
        return()
    endif()

    if(item MATCHES "^\\$<LINK_ONLY:([^>]+)>$")
        append_link_item("${CMAKE_MATCH_1}")
        return()
    endif()

    if(TARGET "${item}")
        get_property(visited GLOBAL PROPERTY visited_link_targets)
        if(item IN_LIST visited)
            return()
        endif()

        list(APPEND visited "${item}")
        set_property(GLOBAL PROPERTY visited_link_targets "${visited}")

        get_target_property(location "${item}" IMPORTED_LOCATION_RELEASE)
        if(NOT location)
            get_target_property(location "${item}" IMPORTED_LOCATION)
        endif()
        if(location)
            append_link_flag("${location}")
        endif()

        get_target_property(target_libraries "${item}" INTERFACE_LINK_LIBRARIES)
        if(target_libraries)
            foreach(target_library IN LISTS target_libraries)
                append_link_item("${target_library}")
            endforeach()
        endif()

        get_target_property(target_options "${item}" INTERFACE_LINK_OPTIONS)
        if(target_options)
            foreach(target_option IN LISTS target_options)
                append_link_flag("${target_option}")
            endforeach()
        endif()
        return()
    endif()

    if(item MATCHES "^-" OR IS_ABSOLUTE "${item}")
        append_link_flag("${item}")
    else()
        append_link_flag("-l${item}")
    endif()
endfunction()

get_target_property(link_libraries SDL3::SDL3-static INTERFACE_LINK_LIBRARIES)
foreach(library IN LISTS link_libraries)
    append_link_item("${library}")
endforeach()

get_target_property(link_options SDL3::SDL3-static INTERFACE_LINK_OPTIONS)
if(link_options)
    foreach(link_option IN LISTS link_options)
        append_link_flag("${link_option}")
    endforeach()
endif()

get_property(linker_flags GLOBAL PROPERTY collected_linker_flags)

list(JOIN compiler_flags "\n" compiler_flags_text)
list(JOIN linker_flags "\n" linker_flags_text)
file(MAKE_DIRECTORY "${OUTPUT_DIRECTORY}")
file(WRITE "${OUTPUT_DIRECTORY}/compiler-flags.txt" "${compiler_flags_text}\n")
file(WRITE "${OUTPUT_DIRECTORY}/linker-flags.txt" "${linker_flags_text}\n")
